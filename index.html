// HÀM XỬ LÝ QR & TRUY VẤN CSDL ĐÃ SỬA LỖI MẤT LỚP async function processQRCode(qrRawText) { if (!qrRawText) return; // 1. Ép kiểu chuỗi và xóa sạch khoảng trắng / dấu ngoặc thừa let cleanCode = String(qrRawText).trim().replace(/^["']|["']$/g, ''); let hsData = null; let sourceName = "Chưa tìm thấy"; // TH1: Chuỗi QR dạng JSON (nếu QR chứa thông tin đóng gói) try { const parsed = JSON.parse(cleanCode); if (parsed.ma_hs || parsed.id) { cleanCode = String(parsed.ma_hs || parsed.id).trim(); if (parsed.lop) { hsData = { ma_hs: cleanCode, ho_ten: parsed.ho_ten || "Học sinh", khoi: parsed.khoi || "Chưa rõ khối", lop: parsed.lop }; sourceName = "Mã QR JSON"; } } } catch (e) {} // TH2: TRUY VẤN CSDL SUPABASE (BẢNG danh_sach) if (!hsData && _supabase) { try { console.log("Đang tìm ma_hs dạng chuỗi:", cleanCode); // Chọn chính xác 4 cột dữ liệu const { data, error } = await _supabase .from('danh_sach') .select('ma_hs, ho_ten, khoi, lop') .or(`ma_hs.eq.${cleanCode},id.eq.${cleanCode}`) .maybeSingle(); if (error) { console.error("Lỗi truy vấn Supabase:", error.message); } else if (data) { console.log("Dữ liệu lấy từ Supabase:", data); hsData = { ma_hs: String(data.ma_hs || cleanCode), ho_ten: data.ho_ten ? String(data.ho_ten).trim() : "Chưa có tên", khoi: data.khoi ? String(data.khoi).trim() : "Chưa rõ khối", lop: data.lop ? String(data.lop).trim() : "Chưa phân lớp" // Đảm bảo đọc cột 'lop' }; sourceName = "CSDL Supabase (Bảng danh_sach)"; } } catch (err) { console.error("Lỗi kết nối máy chủ Supabase:", err); } } // TH3: Tra cứu danh sách Mẫu Cục Bộ (danhSachLocal) if (!hsData) { const foundLocal = danhSachLocal.find(item => String(item.ma_hs) === cleanCode || String(item.id) === cleanCode); if (foundLocal) { hsData = { ma_hs: String(foundLocal.ma_hs), ho_ten: foundLocal.ho_ten, khoi: foundLocal.khoi, lop: foundLocal.lop }; sourceName = "Dữ liệu Mẫu Cục Bộ"; } } // TH4: Nếu KHÔNG tìm thấy ở đâu (Bỏ gán cứng "Khối 11" để tránh báo giả) if (!hsData) { hsData = { ma_hs: cleanCode, ho_ten: `Mã chưa có trong CSDL (${cleanCode})`, khoi: "Chưa rõ khối", lop: "Chưa phân lớp" }; sourceName = "Không tìm thấy trong CSDL"; } // Xử lý hình thức kỷ luật / thưởng const actionEl = document.querySelector('input[name="qr-action"]:checked'); const actionValue = actionEl ? actionEl.value : 'muon_khong_phep'; let actionLabel = "Đi học muộn (Không phép)"; let diemCalc = -3; if (actionValue === 'muon_co_phep') { actionLabel = "Đi học muộn (Có phép)"; diemCalc = -2; } else if (actionValue === 'diem_thuong') { actionLabel = "Khen thưởng (+ Điểm)"; diemCalc = 2; } else if (actionValue === 'diem_tru') { actionLabel = "Vi phạm (- Điểm)"; diemCalc = -2; } currentScannedStudent = { ma_hs: hsData.ma_hs, ho_ten: hsData.ho_ten, khoi: hsData.khoi, lop: hsData.lop, // Thông tin lớp actionValue: actionValue, actionLabel: actionLabel, diem: diemCalc, time: new Date().toLocaleTimeString('vi-VN', { hour: '2-digit', minute: '2-digit', second: '2-digit' }), dateDDMMYYYY: formatDateDDMMYYYY(new Date()), dateYYYYMMDD: formatDateYYYYMMDD(new Date()) }; // Đổ dữ liệu lên màn hình Popup Modal document.getElementById('modal-ma-hs').innerText = currentScannedStudent.ma_hs; document.getElementById('modal-ho-ten').innerText = currentScannedStudent.ho_ten; document.getElementById('modal-khoi').innerText = currentScannedStudent.khoi; document.getElementById('modal-lop').innerText = currentScannedStudent.lop; // <--- Hiển thị Lớp document.getElementById('modal-hinh-thuc').innerText = currentScannedStudent.actionLabel; document.getElementById('source-name-text').innerText = sourceName; const diemText = currentScannedStudent.diem > 0 ? `+${currentScannedStudent.diem} điểm` : `${currentScannedStudent.diem} điểm`; document.getElementById('modal-diem').innerText = diemText; document.getElementById('modal-diem').className = currentScannedStudent.diem > 0 ? "font-bold text-lg text-emerald-600" : "font-bold text-lg text-red-600"; // Hiển thị Modal document.getElementById('qr-modal').classList.remove('hidden'); }
