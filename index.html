// HÀM TRUY VẤN CSDL SUPABASE TỐI ƯU - TỰ ĐỘNG BẮT TÊN CỘT LỚP & KHỐI async function processQRCode(qrRawText) { if (!qrRawText) return; // 1. Làm sạch chuỗi quét đầu vào let cleanCode = String(qrRawText).trim().replace(/^["']|["']$/g, ''); let hsData = null; // TH1: Xử lý nếu mã QR là chuỗi JSON try { const parsed = JSON.parse(cleanCode); if (parsed.ma_hs || parsed.id) { cleanCode = String(parsed.ma_hs || parsed.id).trim(); if (parsed.lop || parsed.ten_lop) { hsData = { ma_hs: cleanCode, ho_ten: parsed.ho_ten || parsed.hoten || "Học sinh", khoi: parsed.khoi || parsed.ten_khoi || "Chưa rõ khối", lop: parsed.lop || parsed.ten_lop || "Chưa phân lớp" }; } } } catch (e) {} // TRUY VẤN CSDL SUPABASE (BƯỚC 1: BẢNG 'danh_sach') if (!hsData && _supabase) { try { console.log("--> Đang tìm mã trong bảng 'danh_sach':", cleanCode); let query = _supabase.from('danh_sach').select('*'); if (!isNaN(cleanCode)) { query = query.or(`ma_hs.eq.${cleanCode},id.eq.${cleanCode}`); } else { query = query.eq('ma_hs', cleanCode); } const { data, error } = await query.maybeSingle(); if (error) { console.warn("Lỗi truy vấn bảng 'danh_sach':", error.message); } else if (data) { console.log("--> Kết quả từ bảng 'danh_sach':", data); hsData = parseStudentRecord(data, cleanCode); } } catch (err) { console.error("Lỗi kết nối bảng 'danh_sach':", err); } } // TRUY VẤN DỰ PHÒNG SUPABASE (BƯỚC 2: BẢNG 'hoc_sinh' NẾU BẢNG 'danh_sach' KHÔNG CÓ) if (!hsData && _supabase) { try { console.log("--> Không thấy ở danh_sach, tiếp tục tìm trong bảng 'hoc_sinh':", cleanCode); let query = _supabase.from('hoc_sinh').select('*'); if (!isNaN(cleanCode)) { query = query.or(`ma_hs.eq.${cleanCode},id.eq.${cleanCode}`); } else { query = query.eq('ma_hs', cleanCode); } const { data, error } = await query.maybeSingle(); if (error) { console.warn("Lỗi truy vấn bảng 'hoc_sinh':", error.message); } else if (data) { console.log("--> Kết quả từ bảng 'hoc_sinh':", data); hsData = parseStudentRecord(data, cleanCode); } } catch (err) { console.error("Lỗi kết nối bảng 'hoc_sinh':", err); } } // TH3: Không tìm thấy dữ liệu ở bất kỳ đâu if (!hsData) { console.warn("Không tìm thấy thông tin học sinh cho mã:", cleanCode); hsData = { ma_hs: cleanCode, ho_ten: `Chưa có thông tin (${cleanCode})`, khoi: "Chưa rõ khối", lop: "Chưa phân lớp" }; } // Lấy hình thức ghi nhận const actionEl = document.querySelector('input[name="qr-action"]:checked'); const actionValue = actionEl ? actionEl.value : 'muon_khong_phep'; let actionLabel = "Đi học muộn (Không phép)"; let maHD = "HD02a"; let diemCalc = -1.0; if (actionValue === 'muon_co_phep') { actionLabel = "Đi học muộn (Có phép)"; maHD = "HD02b"; diemCalc = -0.5; } else if (actionValue === 'diem_thuong') { actionLabel = "Khen thưởng (+ Điểm)"; maHD = "LINH_HOAT"; diemCalc = 2.0; } else if (actionValue === 'diem_tru') { actionLabel = "Vi phạm (- Điểm)"; maHD = "HD21d"; diemCalc = -2.0; } const now = new Date(); const timeString = now.toLocaleTimeString('vi-VN', { hour: '2-digit', minute: '2-digit' }); currentScannedStudent = { ma_hs: hsData.ma_hs, ho_ten: hsData.ho_ten, khoi: hsData.khoi, lop: hsData.lop, actionValue: actionValue, actionLabel: `${actionLabel} (${timeString})`, ma_hd: maHD, buoi: now.getHours() < 12 ? "Sáng" : "Chiều", diem: diemCalc, time: timeString, dateYYYYMMDD: formatDateYYYYMMDD(now) }; // Đổ dữ liệu lên Modal Popup document.getElementById('modal-ma-hs').innerText = currentScannedStudent.ma_hs; document.getElementById('modal-ho-ten').innerText = currentScannedStudent.ho_ten; document.getElementById('modal-khoi').innerText = currentScannedStudent.khoi; document.getElementById('modal-lop').innerText = currentScannedStudent.lop; document.getElementById('modal-hinh-thuc').innerText = currentScannedStudent.actionLabel; const diemText = currentScannedStudent.diem > 0 ? `+${currentScannedStudent.diem} điểm` : `${currentScannedStudent.diem} điểm`; document.getElementById('modal-diem').innerText = diemText; document.getElementById('modal-diem').className = currentScannedStudent.diem > 0 ? "font-bold text-lg text-emerald-600" : "font-bold text-lg text-red-600"; document.getElementById('qr-modal').classList.remove('hidden'); } // HÀM BÓC TÁCH DỮ LIỆU TỰ ĐỘNG DỰA TRÊN MỌI TÊN CỘT CÓ THỂ CÓ TRONG CSDL function parseStudentRecord(data, defaultCode) { if (!data) return null; // 1. Lấy thông tin Lớp (thử lần lượt các tên cột có thể tồn tại) const rawLop = data.lop || data.ten_lop || data.lop_hoc || data.tenlop || data.class || "Chưa phân lớp"; // 2. Lấy thông tin Khối const rawKhoi = data.khoi || data.ten_khoi || data.khoi_hoc || data.tenkhoi || data.grade || "Chưa rõ khối"; // 3. Lấy Họ và tên const rawHoTen = data.ho_ten || data.hoten || data.ten_hoc_sinh || data.ten_hs || data.full_name || "Chưa có tên"; // 4. Lấy Mã HS const rawMaHS = data.ma_hs || data.ma_hoc_sinh || data.id || defaultCode; return { ma_hs: String(rawMaHS).trim(), ho_ten: String(rawHoTen).trim(), khoi: String(rawKhoi).trim(), lop: String(rawLop).trim() }; }
